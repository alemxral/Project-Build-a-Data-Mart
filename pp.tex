\section{Testing and Integrity Checks}
\begin{frame}[fragile]{Sample Integrity Test}
  \textbf{Join Test:}
  \begin{lstlisting}[language=SQL]
SELECT r.reservation_id,
       CONCAT(up.first_name,' ',up.last_name) AS guest,
       p.title AS property,
       t.amount AS paid
FROM reservations r
JOIN guests g ON g.guest_id = r.guest_id
JOIN user_profiles up ON up.user_id = g.user_id
JOIN property_listings p ON p.property_id = r.property_id
JOIN transactions t ON t.reservation_id = r.reservation_id
LIMIT 5;
  \end{lstlisting}
